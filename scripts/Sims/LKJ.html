library(cmdstanr)
library(posterior)
library(dplyr)
library(tidyr)
library(ggplot2)
library(patchwork)
set_cmdstan_path("C:/Users/pawel/Documents/.cmdstan/Torsten-torsten_v0.92.0/Torsten-torsten_v0.92.0/cmdstan")

stan_code <- '
data {
  int<lower=1> N;
  real<lower=0> eta;
}

generated quantities {
  vector[N] rho12;
  vector[N] rho13;
  vector[N] rho23;

  for (n in 1:N) {
    matrix[8,8] R;
    R = lkj_corr_rng(8, eta);

    rho12[n] = R[1,2];
    rho13[n] = R[1,3];
    rho23[n] = R[2,3];
  }
}
'

mod <- cmdstan_model(
  write_stan_file(stan_code)
)

simulate_lkj <- function(eta, N = 2000) {

  fit <- mod$sample(
    data = list(N = N, eta = eta),
    chains = 1,
    iter_warmup = 0,
    iter_sampling = 1,
    fixed_param = TRUE,
    refresh = 0
  )

  fit$draws(
    variables = c("rho12", "rho13", "rho23"),
    format = "draws_df"
  ) |>
    as.data.frame() |>
    mutate(eta = paste0("LKJ(", eta, ")"))
}

set.seed(123)

draws <- bind_rows(
  simulate_lkj(1),
  simulate_lkj(10)
)
